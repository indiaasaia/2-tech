%
% cover_letter_template_de.tex — German Anschreiben, DIN 5008-style one-pager
%
% Structure per the German norm (Agentur für Arbeit "Das 1x1 der Bewerbungsunterlagen"):
%   sender block top-right, company block left, city + DD.MM.YYYY date right,
%   BOLD subject line with the exact job title (no "Betreff:" word), gendered
%   salutation, 3-4 short paragraphs, "Mit freundlichen Grüßen" + signature.
%
% Parser-safety: the V3/V4-validated recipe (texgyrepagella with
% Ligatures=NoCommon, four-digit dates, \bfseries titles, collision-proof
% at-token filling, longest-first replacement + leftover-assert in fill.py).
% English/EU regexes must still extract the name, email, phone and dates from
% the compiled PDF.
%
\documentclass[a4paper,11pt]{article}
\usepackage[empty]{fullpage}
\usepackage{geometry}
\usepackage{xcolor}
\usepackage{parskip}
\usepackage[hidelinks]{hyperref}

\geometry{left=2.4cm, right=2cm, top=2.2cm, bottom=2cm}
\pagestyle{empty}
\setlength{\parskip}{0.45em}
\setlength{\parindent}{0pt}

% Parser-safety: FLATTERSATZ (ragged right) — justified text compresses
% interword spaces below pdfplumber's x-tolerance and the extracted text
% loses ALL word boundaries ("MitfreundlichenGrüßen", measured in E2E).
% The parser-safe resume template makes the same choice.
\raggedright

% Parser-safety: widen the interword glue. TeX Gyre Pagella's 0.25em space
% yields ~2.7pt x-gaps at 10pt — below pdfplumber's default x_tolerance of 3 —
% so extractors merge every word ("MitfreundlichenGrüßen", measured E2E).
% \spaceskip forces a >= 3.3pt stretchable space; at-shintos keep raggedness.
\setlength{\spaceskip}{0.34em plus 0.14em minus 0.06em}
\setlength{\xspaceskip}{0.34em plus 0.14em minus 0.06em}

\usepackage{fontspec}
\setmainfont{texgyrepagella}[Ligatures=NoCommon, Extension=.otf, UprightFont=*-regular, BoldFont=*-bold, ItalicFont=*-italic, BoldItalicFont=*-bolditalic]

\newcommand{\clsender}[1]{{\small #1\par}}
